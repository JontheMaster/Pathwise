Dieselbe Flutter-Oberfläche im Handy-Browser, als installierte App auf iOS und Android, auf dem Tablet und im Desktop-Browser. Mobil ist die Wahrheit: eine 390er-Spalte, 44px-Bedienelemente, die Beratungsleiste immer am Fuß. Ab 900px kommt links die Szenarienliste dazu, die Spalte selbst bleibt gleich. router.dart · go_router 
// Ein Pfad je Bildschirm, damit der Link im Vereinschat
// direkt in ein Szenario führen kann. Kein Login, keine Guards.
final router = GoRouter(
  initialLocation: '/szenarien',
  routes: [
    ShellRoute(
      builder: (c, s, child) => PathwiseShell(child: child),
      routes: [
        GoRoute(path: '/szenarien',
          builder: (c, s) => const UebersichtPage()),
        GoRoute(path: '/szenario/:id',
          builder: (c, s) => EinstiegPage(id: s.pathParameters['id']!),
          routes: [
            // Entscheidungspunkt 1..n — Zurück geht einen Punkt zurück,
            // nicht aus dem Szenario heraus.
            GoRoute(path: 'punkt/:nr',
              builder: (c, s) => PunktPage(
                id: s.pathParameters['id']!,
                nr: int.parse(s.pathParameters['nr']!),
              )),
            GoRoute(path: 'auswertung',
              builder: (c, s) => AuswertungPage(
                id: s.pathParameters['id']!)),
          ]),
        GoRoute(path: '/verein',
          builder: (c, s) => const VereinPage()),
      ],
    ),
  ],
);

// Hilfe und Beratung ist keine Route, sondern ein Overlay:
// der Durchlauf darf dabei nicht verloren gehen.
void openHilfe(BuildContext c) => MediaQuery.sizeOf(c).width < 900
    ? showModalBottomSheet(context: c, builder: (_) => const HilfeSheet())
    : showDialog(context: c, builder: (_) => const HilfeDialog());
